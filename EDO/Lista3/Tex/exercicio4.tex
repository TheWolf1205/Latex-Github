\begin{homeworkProblem}[4]
  Considere o sistema
  \begin{align*}
    X^{\prime} = \begin{pmatrix}
      1 & \alpha\\
      1 & 1
      \end{pmatrix}X,\qquad \alpha\in\mathbb{R}.
  \end{align*}
  \begin{enumerate}
    \item Determine os valores de $\alpha$ para os quais o sistema é
    topologicamente conjugado a
    \begin{align*}
      X' = \begin{pmatrix}
        1 & 0\\
        1 & 1
      \end{pmatrix}X.
    \end{align*}
    \item Determine os valores de $\alpha$ para os quais o sistema é
    topologicamente conjugado a
    \begin{align*}
      X^{\prime} = \begin{pmatrix}
        4 & 0\\
        0 & -2
      \end{pmatrix}X.
    \end{align*}
  \end{enumerate}
  \begin{solution}
  \begin{enumerate}
    \item Note que a matriz tem autovalor positivo $\lambda=1$ de multiplicidade $2$, pelo que vamos procurar que $\alpha$ satisfaz que $\nu=1\pm\sqrt{\alpha}$ seja positivo nos dois casos. Obtendo que $0\leq \alpha<1$. (Caso contrário o diagrama de fase troca e portanto não é topológicamente conjugado).
    \item Note que a matríz tem autovalores $\lambda=4$ e $\lambda=-2$, isto é, autovalores reais de sinal contrário. Isto implica que $\alpha>1$, caso contrário $\nu$ é complexo o de multiplicidade $2$. 
  \end{enumerate} 
  \end{solution}
\end{homeworkProblem}
